\begin{tikzpicture}[

    %---------- AP location ----------

    \filldraw[fill=gray!28, draw=black, line width=0.5pt]

    \draw[line width=0.5pt, dashdotted, black!70] (AP) -- ++(0:8.3);

        ($(AP)+(13:1.6)$) arc[start angle=13, end angle=-13, radius=1.6];
    \node[anchor=west, font=\scriptsize] at ($(AP)+(0:1.7)$) {$\theta_{\text{HPBW}}$};

    \node[anchor=west, font=\scriptsize, text=cb_orange!85!black]

        ($(AP)+(-10:7.7)$) .. controls (8.0,-1.55) and (7.85,-1.65) .. (TrigLab.west);

        {$\|\dot{\boldsymbol{\delta}}\|\,T_{\text{react}}$};

        {UE @ $t + T_{\text{react}}$};
