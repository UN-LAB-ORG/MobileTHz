% trade-off. (Reviewer 1 R1.2 / Editor #2).

    \begin{axis}[
        width=\linewidth, height=5.2cm,

            legend columns=1,
            legend pos=south west,
            inner sep=1pt,
            row sep=0pt,
            column sep=2pt,
        },
        grid=major,
        grid style={dashed, gray!50},
    ]

    \end{axis}

    % Right-hand axis: signaling overhead
    \begin{axis}[
